"""Minimal ISO9660 extractor (little-endian records) to pull out boot files."""
import struct, os, sys

def parse_dir_records(buf):
    recs = []
    off = 0
    n = len(buf)
    while off < n:
        reclen = buf[off]
        if reclen == 0:
            off += 1
            continue
        if off + reclen > n:
            break
        rec = buf[off:off+reclen]
        if reclen < 33:
            off += reclen
            continue
        flags = rec[17]
        name_len = rec[32]
        if 33 + name_len > reclen:
            off += reclen
            continue
        name = rec[33:33+name_len].decode('latin1')
        extent = struct.unpack('<I', rec[2:6])[0]
        length = struct.unpack('<I', rec[6:10])[0]
        if name in ('.', '..'):
            off += reclen
            continue
        recs.append((name, extent, length, bool(flags & 2)))
        off += reclen
    return recs

def read_dir(extent):
    return data[extent*SEC:(extent+1)*SEC]

def find_in_tree(start_extent, target, want_dir):

        if ext in seen:

        seen.add(ext)
        buf = read_dir(ext)
        for name, e, l, isdir in parse_dir_records(buf):
            if name == target and isdir == want_dir:
                return e, l, isdir
            if isdir:
